1Objetivo

Comparar as tabelas da camada Bronze (voebem.bronze) com suas versões transformadas na camada Silver (voebem.silver) para demonstrar, na prática, como a transformação altera estrutura, tipos e qualidade de uso dos dados.

Escopo seguro: todas as consultas do notebook são exclusivamente de leitura. Nenhum dado foi inserido, atualizado, excluído ou modificado.
Somente leituraPySpark + SQLDelta / DatabricksArquitetura Medalhão

2Contexto do Pipeline

Fluxo analisado

voebem.bronze→voebem.silver

A Bronze preserva o dado bruto e, como regra, mantém os campos de negócio em STRING. A Silver aplica tipagem semântica, padronização de valores, nomes mais claros, documentação e colunas derivadas para consumo analítico.

  • Comparação de contagem para verificar preservação de linhas.
  • Comparação coluna a coluna via information_schema.
  • Validação de 'null', 'N/A', vírgula decimal e renomeações.
  • Inspeção dos mesmos tipos de registros antes e depois da transformação.

Mapeamento Bronze → Silver

Resultado: cinco tabelas físicas da Bronze são comparadas com quatro tabelas da Silver; as duas origens de empresas são unificadas com rastreabilidade por origem_cadastro.

display(spark.sql("""
    SELECT 'vra' AS tabela,
           (SELECT COUNT(*) FROM voebem.bronze.vra) AS bronze,
           (SELECT COUNT(*) FROM voebem.silver.vra) AS silver,
           (SELECT COUNT(*) FROM voebem.bronze.vra) - (SELECT COUNT(*) FROM voebem.silver.vra) AS diferenca
    UNION ALL
    SELECT 'aerodromos',
           (SELECT COUNT(*) FROM voebem.bronze.aerodromos),
           (SELECT COUNT(*) FROM voebem.silver.aerodromos),
           (SELECT COUNT(*) FROM voebem.bronze.aerodromos) - (SELECT COUNT(*) FROM voebem.silver.aerodromos)
    UNION ALL
    SELECT 'codigos_operacao',
           (SELECT COUNT(*) FROM voebem.bronze.codigos_operacao),
           (SELECT COUNT(*) FROM voebem.silver.codigos_operacao),
           (SELECT COUNT(*) FROM voebem.bronze.codigos_operacao) - (SELECT COUNT(*) FROM voebem.silver.codigos_operacao)
    UNION ALL
    SELECT 'empresas (nac + est)',
           (SELECT COUNT(*) FROM voebem.bronze.empresas_nacionais) + (SELECT COUNT(*) FROM voebem.bronze.empresas_estrangeiras),
           (SELECT COUNT(*) FROM voebem.silver.empresas),
           (SELECT COUNT(*) FROM voebem.bronze.empresas_nacionais) + (SELECT COUNT(*) FROM voebem.bronze.empresas_estrangeiras) - (SELECT COUNT(*) FROM voebem.silver.empresas)
"""))
Exibir código completo — VRA: comparação de tipos bronze vs silver2.230 s
display(spark.sql("""
    SELECT
        'vra'                     AS tabela,
        COALESCE(b.column_name, s.column_name) AS coluna,
        b.data_type               AS tipo_bronze,
        s.data_type               AS tipo_silver,
        CASE
            WHEN b.column_name IS NULL THEN 'NOVA (criada na silver)'
            WHEN s.column_name IS NULL THEN 'REMOVIDA'
            WHEN b.data_type <> s.data_type THEN CONCAT('TIPO MUDOU: ', b.data_type, ' -> ', s.data_type)
            ELSE 'igual'
        END                       AS observacao
    FROM (
        SELECT column_name, data_type
        FROM voebem.information_schema.columns
        WHERE table_schema = 'bronze' AND table_name = 'vra'
    ) b
    FULL OUTER JOIN (
        SELECT column_name, data_type
        FROM voebem.information_schema.columns
        WHERE table_schema = 'silver' AND table_name = 'vra'
    ) s ON b.column_name = s.column_name
    ORDER BY
        CASE WHEN b.column_name IS NULL THEN 1 ELSE 0 END,
        COALESCE(b.column_name, s.column_name)
"""))

Exibir código completo — Empresas: silver vs bronze (nacionais + estrangeiras)1.922 s
display(spark.sql("""
    WITH silver_cols AS (
        SELECT column_name, data_type
        FROM voebem.information_schema.columns
        WHERE table_schema = 'silver' AND table_name = 'empresas'
    ),
    bronze_nac AS (
        SELECT column_name, data_type
        FROM voebem.information_schema.columns
        WHERE table_schema = 'bronze' AND table_name = 'empresas_nacionais'
    ),
    bronze_est AS (
        SELECT column_name, data_type
        FROM voebem.information_schema.columns
        WHERE table_schema = 'bronze' AND table_name = 'empresas_estrangeiras'
    )
    SELECT
        'empresas'               AS tabela,
        s.column_name            AS coluna_silver,
        s.data_type              AS tipo_silver,
        bn.data_type             AS tipo_bronze_nacionais,
        be.data_type             AS tipo_bronze_estrangeiras,
        CASE
            WHEN bn.column_name IS NULL AND be.column_name IS NULL THEN 'NOVA (criada na silver)'
            WHEN bn.column_name IS NOT NULL AND be.column_name IS NULL THEN 'so nacionais'
            WHEN bn.column_name IS NULL AND be.column_name IS NOT NULL THEN 'so estrangeiras'
            ELSE 'herdada de ambas'
        END                      AS observacao
    FROM silver_cols s
    LEFT JOIN bronze_nac bn ON s.column_name = bn.column_name
    LEFT JOIN bronze_est be ON s.column_name = be.column_name
    ORDER BY s.column_name
"""))
Exibir código completo — VRA: string 'null' e 'N/A' vs NULL real2.566 s
display(spark.sql("""
    SELECT
        'bronze.vra'                                                        AS tabela,
        SUM(CASE WHEN partida_real = 'null'           THEN 1 ELSE 0 END)      AS partida_real_str_null,
        SUM(CASE WHEN partida_real IS NULL            THEN 1 ELSE 0 END)      AS partida_real_null_real,
        SUM(CASE WHEN chegada_real = 'null'            THEN 1 ELSE 0 END)      AS chegada_real_str_null,
        SUM(CASE WHEN chegada_real IS NULL             THEN 1 ELSE 0 END)      AS chegada_real_null_real,
        SUM(CASE WHEN codigo_justificativa = 'N/A'     THEN 1 ELSE 0 END)      AS justificativa_str_NA,
        COUNT(*)                                                           AS total_linhas
    FROM voebem.bronze.vra
    UNION ALL
    SELECT
        'silver.vra',
        0,
        SUM(CASE WHEN partida_real IS NULL THEN 1 ELSE 0 END),
        0,
        SUM(CASE WHEN chegada_real IS NULL THEN 1 ELSE 0 END),
        0,
        COUNT(*)
    FROM voebem.silver.vra
"""))

    SELECT
        b.icao,
        b.altitude        AS altitude_bronze,
